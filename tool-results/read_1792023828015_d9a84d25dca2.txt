   241→/** Plain-text posting section rendered as a disc bullet list. */
   242→function TextBullets({ text }: { text: string }) {
   243→  const items = text
   244→    .split(/\r?\n/)
   245→    .map((l) => l.replace(/^[-•*]\s*/, "").trim())
   246→    .filter(Boolean);
   247→  if (items.length === 0) return null;
   248→  return (
   249→    <ul className="list-disc space-y-2 pl-5 text-base leading-relaxed text-stone">
   250→      {items.map((item, i) => (
   251→        <li key={i}>{item}</li>
   252→      ))}
   253→    </ul>
   254→  );
   255→}
   256→
   257→/** Posting section: sanitized rich HTML when present, else text bullets. */
   258→function PostingSection({ title, html, text }: { title: string; html: string | null; text: string | null }) {
   259→  if (html && html.trim()) {
   260→    return (
   261→      <section className="space-y-3">
   262→        <h3 className="text-lg font-medium text-ink">{title}</h3>
   263→        <div className="rich-text text-base leading-relaxed text-stone" dangerouslySetInnerHTML={{ __html: html }} />
   264→      </section>
   265→    );
   266→  }
   267→  if (text && text.trim()) {
   268→    return (
   269→      <section className="space-y-3">
   270→        <h3 className="text-lg font-medium text-ink">{title}</h3>
   271→        <TextBullets text={text} />
   272→      </section>
   273→    );
   274→  }
   275→  return null;
   276→}

   278→function JobDetail({
   279→  job,
   280→  saved,
   281→  onApply,
   282→  onSave,
   283→  onBack,
   284→}: {
   285→  job: JobWire;
   286→  saved: boolean;
   287→  onApply: () => void;
   288→  onSave: () => void;
   289→  onBack: () => void;
   290→}) {
   291→  const pos = job.position;
   292→  const title = pos?.positionTitle || job.title || "Position Title Unavailable";
   293→  const salary = pos?.salaryAmount ?? null;
   294→  const deadline = job.deadlineDate;
   295→  const overdue = jobDeadlinePassed(deadline);
   296→
   297→  const meta: { icon: typeof MapPin; label: string; num?: boolean; danger?: boolean }[] = [];
   298→  // MOM 2026-10-13 — division meta rows read the full updated division name.
   299→  if (job.position?.division) meta.push({ icon: Building2, label: divisionFullName(job.position.division) });
   300→  if (job.positionType) meta.push({ icon: Briefcase, label: humanize(job.positionType) });
   301→  if (salary != null)
   302→    meta.push({
   303→      icon: Banknote,
   304→      label: `${formatCurrency(salary)} per month${
   305→        pos?.salaryGrade ? ` · SG ${pos.salaryGrade}${pos.salaryStep ? `/${pos.salaryStep}` : ""}` : ""
   306→      }`,
   307→      num: true,
   308→    });
   309→  meta.push({
   310→    icon: Clock,
   311→    label: overdue ? "Closed: deadline passed" : deadline ? `Closes ${formatDate(deadline)}` : "Open until filled",
   312→    num: !overdue && !deadline,
   313→    danger: overdue,
   314→  });
   315→
   316→  return (
   317→    <div>
   318→      {/* Mobile — back to the register */}
   319→      <button
   320→        type="button"
   321→        onClick={onBack}
   322→        className="focus-ring mb-4 inline-flex items-center gap-1.5 text-sm font-medium text-metric-blue lg:hidden"
   323→      >
   324→        <ChevronLeft className="h-4 w-4" aria-hidden="true" /> Back to job list
   325→      </button>
   326→
   327→      {/* Posting title — the house page-title step (24px). The pane is a
   328→          document view: it must sit below the 44px hero, not compete. */}
   329→      <h2 className="text-2xl font-medium leading-tight text-ink">
   330→        {humanize(title)}
   331→      </h2>
   332→      {/* MOM 2026-10-13 — "Remove DOST-MIRDC": the agency line (and its
   333→          verified badge) is gone from the posting header — the issuing
   334→          agency is the site itself; only the register link remains. */}
   335→      <div className="mt-2.5 flex flex-wrap items-center gap-x-2.5 gap-y-1.5">
   336→        <button
   337→          type="button"
   338→          onClick={() => navigate("vacant-positions")}
   339→          className="focus-ring text-[15px] text-metric-blue underline underline-offset-2 transition-colors hover:text-ink"
   340→        >
   341→          View all jobs
   342→        </button>
   343→      </div>
   344→
   345→      <ul className="mt-4 space-y-2.5">
   346→        <li className="flex items-start gap-2.5 text-base text-stone">
   347→          <MapPin className="mt-0.5 h-[18px] w-[18px] shrink-0 text-stone" aria-hidden="true" />
   348→          {jobPlace(job)}
   349→        </li>
   350→        {meta.map((m, i) => (
   351→          <li
   352→            key={i}
   353→            className={cn(
   354→              "flex items-start gap-2.5 text-base",
   355→              m.danger ? "font-medium text-[var(--bad)]" : "text-stone"
   356→            )}
   357→          >
   358→            <m.icon className="mt-0.5 h-[18px] w-[18px] shrink-0 text-stone" aria-hidden="true" />
   359→            <span className={cn(m.num && "num")}>{m.label}</span>
   360→          </li>
   361→        ))}
   362→      </ul>
   363→
   364→      <p className="num mt-4 text-sm text-stone">Posted {timeAgo(postedAt(job))}</p>
   365→
   366→      <div className="mt-5 flex flex-wrap items-center gap-3">
   367→        <button type="button" onClick={onApply} disabled={overdue} className={ctaBtn}>
   368→          Quick apply
   369→        </button>
   370→        <button
   371→          type="button"
   372→          onClick={onSave}
   373→          aria-pressed={saved}
   374→          className={ghostBtn}
   375→        >
   376→          <Bookmark className={cn("h-4 w-4", saved && "fill-current")} aria-hidden="true" />
   377→          {saved ? "Saved" : "Save"}
   378→        </button>
   379→        {overdue && (
   380→          <span className="flex items-center gap-1.5 text-xs font-medium text-[var(--bad)]">
   381→            <AlertCircle className="h-3.5 w-3.5" aria-hidden="true" /> Deadline passed
   382→          </span>
   383→        )}
   384→      </div>
   385→
   386→      {/* Posting body */}
   387→      <div className="mt-8 space-y-8 border-t border-divider pt-8">
   388→        {job.briefDescription &&
   389→          job.briefDescription
   390→            .split(/\n{2,}/)
   391→            .filter((p) => p.trim())
   392→            .map((p, i) => (
   393→              <p key={i} className="text-base leading-relaxed text-stone">
   394→                {p}
   395→              </p>
   396→            ))}
   397→        <PostingSection title="Responsibilities" html={job.dutiesHtml} text={job.dutiesResponsibilities} />
   398→        <PostingSection
   399→          title="Qualifications"
   400→          html={job.otherQualificationsHtml}
   401→          text={job.otherQualifications}
   402→        />
   403→        <PostingSection title="Compensation" html={job.compensationHtml} text={job.compensationPackage} />
   404→      </div>
   405→    </div>
   406→  );
   407→}

   411→export default function PublicLanding() {
   412→  const [jobs, setJobs] = useState<JobWire[] | null>(null);
   413→  const [error, setError] = useState<string | null>(null);
   414→  // Last non-null error — keeps the BEUI alert's exit animation populated with
   415→  // content while the message state has already cleared.
   416→  const [lastError, setLastError] = useState<string | null>(null);
   417→  useEffect(() => {
   418→    if (error !== null) setLastError(error);
   419→  }, [error]);
   420→  const [selectedId, setSelectedId] = useState<number | null>(null);
   421→  const [expandedId, setExpandedId] = useState<number | null>(null);
   422→  const [isDesktop, setIsDesktop] = useState(false);
   423→  const lastJson = useRef("");
   424→  const paneRef = useRef<HTMLDivElement>(null);
   425→  const { saved, toggle: toggleSave } = useSavedJobs();
   426→  // Shared apply gate — anonymous visitors get the sign-in prompt, complete
   427→  // applicants get the confirm flow (same as the board), incomplete profiles
   428→  // are sent to finish their profile first (the fast-track was retired).
   429→  const { requestApply, dialogs } = useApplyFlow();
   430→
   431→  const load = useCallback(async (silent: boolean) => {
   432→    try {
   433→      const data = await apiFetch<JobWire[]>("/api/jobs?limit=200");
   434→      const json = JSON.stringify(data);
   435→      if (!silent || json !== lastJson.current) {
   436→        lastJson.current = json;
   437→        setJobs(data);
   438→      }
   439→      setError(null);
   440→    } catch (e) {
   441→      if (!silent) setError(e instanceof Error ? e.message : "Failed to load positions");
   442→    }
   443→  }, []);
   444→
   445→  // Mount + 30s silent poll + refetch on focus (spec §13 cadence).
   446→  useEffect(() => {
   447→    void load(false);
   448→    const timer = setInterval(() => {
   449→      if (!document.hidden) void load(true);
   450→    }, 30_000);
   451→    const onFocus = () => void load(true);
   452→    window.addEventListener("focus", onFocus);
   453→    return () => {
   454→      clearInterval(timer);
   455→      window.removeEventListener("focus", onFocus);
   456→    };
   457→  }, [load]);
   458→
   459→  // Breakpoint sync — desktop keeps both panes; mobile swaps list ↔ detail.
   460→  useEffect(() => {
   461→    const mq = window.matchMedia("(min-width: 1024px)");
   462→    const sync = () => setIsDesktop(mq.matches);
   463→    sync();
   464→    mq.addEventListener("change", sync);
   465→    return () => mq.removeEventListener("change", sync);
   466→  }, []);
   467→
   468→  const openJobs = useMemo(
   469→    () => (jobs ?? []).filter((j) => !deadlineState(j.deadlineDate).overdue),
   470→    [jobs]
   471→  );
   472→
   473→  const selectedJob = useMemo(
   474→    () => openJobs.find((j) => j.id === selectedId) ?? null,
   475→    [openJobs, selectedId]
   476→  );
   477→
   478→  // Desktop opens with the first posting pre-selected (reference-board
   479→  // behavior); a stale selection after a poll snaps back to the top card.
   480→  useEffect(() => {
   481→    if (isDesktop && selectedId === null && openJobs.length > 0) {
   482→      setSelectedId(openJobs[0].id);
   483→    }
   484→  }, [isDesktop, openJobs, selectedId]);
   485→
   486→  useEffect(() => {
   487→    if (selectedId !== null && !openJobs.some((j) => j.id === selectedId)) {
   488→      setSelectedId(openJobs[0]?.id ?? null);
   489→    }
   490→  }, [openJobs, selectedId]);
   491→
   492→  // Reference-board behavior — a freshly picked posting opens from the top,
   493→  // even when the previous posting was scrolled deep into its body.
   494→  useEffect(() => {
   495→    if (paneRef.current) paneRef.current.scrollTop = 0;
   496→  }, [selectedId]);
   497→
   498→  const select = useCallback(
   499→    (id: number) => {
   500→      setSelectedId(id);
   501→      if (!window.matchMedia("(min-width: 1024px)").matches) {
   502→        window.scrollTo({ top: 0, behavior: "smooth" });
   503→      }
   504→    },
   505→    []
   506→  );
   507→
   508→  const saveAndToast = useCallback(
   509→    (job: JobWire) => {
   510→      const wasSaved = saved.includes(job.id);
   511→      toggleSave(job.id);
   512→      toast.success(
   513→        wasSaved
   514→          ? "Removed from your saved jobs"
   515→          : `Saved ${humanize(job.position?.positionTitle || job.title)}`
   516→      );
   517→    },
   518→    [saved, toggleSave]
   519→  );
   520→
   521→  const showList = isDesktop || selectedId === null;
   522→  const showDetail = isDesktop || selectedId !== null;
   523→
