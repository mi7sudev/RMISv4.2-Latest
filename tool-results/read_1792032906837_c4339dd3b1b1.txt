   406→  // ── minimal but VALID one-page PDF generator (document previews render) ──
   407→  function makePdf(title: string, lines: string[]): Buffer {
   408→    const esc = (s: string) => s.replace(/\\/g, "\\\\").replace(/\(/g, "\\(").replace(/\)/g, "\\)");
   409→    let stream = `BT /F1 16 Tf 72 740 Td 22 TL (${esc(title)}) Tj T*\n`;
   410→    for (const l of lines) stream += `(${esc(l)}) Tj T*\n`;
   411→    stream += "ET";
   412→    const objs = [
   413→      "<< /Type /Catalog /Pages 2 0 R >>",
   414→      "<< /Type /Pages /Kids [3 0 R] /Count 1 >>",
   415→      "<< /Type /Page /Parent 2 0 R /MediaBox [0 0 612 792] /Resources << /Font << /F1 5 0 R >> >> /Contents 4 0 R >>",
   416→      `<< /Length ${stream.length} >>\nstream\n${stream}\nendstream`,
   417→      "<< /Type /Font /Subtype /Type1 /BaseFont /Helvetica >>",
   418→    ];
   419→    let pdf = "%PDF-1.4\n";
   420→    const offsets: number[] = [];
   421→    objs.forEach((body, i) => {
   422→      offsets.push(pdf.length);
   423→      pdf += `${i + 1} 0 obj\n${body}\nendobj\n`;
   424→    });
   425→    const xrefStart = pdf.length;
   426→    pdf += `xref\n0 ${objs.length + 1}\n0000000000 65535 f \n`;
   427→    for (const off of offsets) pdf += `${String(off).padStart(10, "0")} 00000 n \n`;
   428→    pdf += `trailer\n<< /Size ${objs.length + 1} /Root 1 0 R >>\nstartxref\n${xrefStart}\n%%EOF`;
   429→    return Buffer.from(pdf, "latin1");
   430→  }

   473→  async function ensureDemoApplicant(p: DemoProfile): Promise<number> {
   474→    let userId: string | null = null;
   475→    if (p.username) {
   476→      const user = await db.user.upsert({

   478→        update: { password: hash, role: "APPLICANT", blocked: false },

   480→      });
   481→      userId = user.id;
   482→    }
   483→    const applicant =
   484→      (userId ? await db.applicant.findUnique({ where: { userId } }) : null) ??
   485→      (await db.applicant.findFirst({ where: { emailAddress: p.email, userId: null } }));
   486→    let row = applicant;
   487→    if (!row) {
   488→      row = await db.applicant.create({
   489→        data: {
   490→          userId, firstName: p.firstName, middleName: p.middleName, lastName: p.lastName,
   491→          emailAddress: p.email, mobileNumber: p.mobile, contactNumber: p.mobile,
   492→          gender: p.gender, civilStatus: p.civilStatus, citizenship: "Filipino",
   493→          birthDate: p.birthDate, birthPlace: p.birthPlace,
   494→          presentAddress: p.street, street: p.street, city: p.city, province: p.province,
   495→          country: "Philippines", zipCode: p.zipCode,
   496→          isProfileComplete: p.submittedDaysAgo != null,
   497→          submittedDate: p.submittedDaysAgo != null ? new Date(now - p.submittedDaysAgo * day) : null,
   498→        },
   499→      });
   500→    }
   501→    if (!row.characterReference && p.refs.length)
   502→      row = await db.applicant.update({ where: { id: row.id }, data: { characterReference: JSON.stringify(p.refs) } });
   503→    // Purge noise rows (no level AND no course/degree/specify) — they can
   504→    // never satisfy an MQR education clause but would block section seeding.
   505→    await db.education.deleteMany({
   506→      where: { applicantId: row.id, AND: [{ educationLevel: null }, { course: null }, { degree: null }, { specifyOthers: null }] },
   507→    });
   508→    // Section fill is idempotent — but a half-empty row (e.g. a leftover
   509→    // all-null entry from manual testing) must NOT block proper seeding, so
   510→    // the counts look for rows with meaningful content only.
   511→    if ((await db.education.count({ where: { applicantId: row.id, OR: [{ educationLevel: { not: null } }, { course: { not: null } }, { degree: { not: null } }, { specifyOthers: { not: null } }] } })) === 0 && p.educations.length)
   512→      await db.education.createMany({ data: p.educations.map((e, i) => ({ applicantId: row!.id, ...e, ord: i + 1 })) });
   513→    if ((await db.workExperience.count({ where: { applicantId: row.id, positionTitle: { not: null } } })) === 0 && p.experiences.length)
   514→      await db.workExperience.createMany({
   515→        data: p.experiences.map((e, i) => ({
   516→          applicantId: row!.id, positionTitle: e.positionTitle, employerName: e.employerName,
   517→          dateFrom: new Date(e.dateFrom), dateTo: e.dateTo ? new Date(e.dateTo) : null,
   518→          isPresentWork: !!e.isPresentWork, isGovtService: !!e.isGovtService,
   519→          statusOfEmployment: e.statusOfEmployment, monthlySalary: e.monthlySalary,
   520→          yearDecimal: e.yearDecimal, ord: i + 1,
   521→        })),
   522→      });
   523→    if ((await db.training.count({ where: { applicantId: row.id, title: { not: null } } })) === 0 && p.trainings.length)
   524→      await db.training.createMany({
   525→        data: p.trainings.map((t, i) => ({
   526→          applicantId: row!.id, title: t.title, typeOfTraining: t.typeOfTraining,
   527→          numberHours: t.numberHours, dateFrom: new Date(t.dateFrom), dateTo: new Date(t.dateTo), ord: i + 1,
   528→        })),
   529→      });
   530→    if ((await db.eligibility.count({ where: { applicantId: row.id } })) === 0 && p.eligibilities.length)
   531→      await db.eligibility.createMany({ data: p.eligibilities.map((e, i) => ({ applicantId: row!.id, ...e, ord: i + 1 })) });
   532→    if ((await db.award.count({ where: { applicantId: row.id } })) === 0 && p.awards?.length)
   533→      await db.award.createMany({ data: p.awards.map((a, i) => ({ applicantId: row!.id, ...a, ord: i + 1 })) });
   534→    return row.id;
   535→  }

   537→  async function ensureDemoApplication(applicantId: number, job: (typeof jobs)[number] | undefined, status: string, daysAgo: number, reason?: string | null) {
   538→    if (!job) return null;
   539→    const rows = await db.application.findMany({ where: { applicantId, jobId: job.id }, orderBy: { dateApplied: "desc" } });
   540→    let row = rows.find((r) => isActiveStatus(r.status)) ?? rows[0] ?? null;
   541→    if (row) {
   542→      if (row.status !== status || (reason ?? null) !== row.reason)
   543→        await db.application.update({ where: { id: row.id }, data: { status, reason: reason ?? null } });
   544→    } else {
   545→      row = await db.application.create({
   546→        data: { applicantId, jobId: job.id, status, reason: reason ?? null, dateApplied: new Date(now - daysAgo * day) },
   547→      });
   548→      await writeSnapshots(row.id, applicantId);
   549→    }
   550→    return row;
   551→  }

   553→  /** The MOM re-application pair: one REJECTED row + one active re-apply row
   554→   *  on the same posting (drives the queue chip, workspace banner, Re-apply). */
   555→  async function ensureReapplyPair(applicantId: number, job: (typeof jobs)[number] | undefined, rejectedDaysAgo: number, reason: string, reappliedDaysAgo: number) {
   556→    if (!job) return;
   557→    const rows = await db.application.findMany({ where: { applicantId, jobId: job.id }, orderBy: { dateApplied: "asc" } });
   558→    let rejected = rows.find((r) => isRejStatus(r.status)) ?? null;
   559→    if (!rejected) {
   560→      rejected = await db.application.create({
   561→        data: { applicantId, jobId: job.id, status: "Rejected", reason, dateApplied: new Date(now - rejectedDaysAgo * day) },
   562→      });
   563→      await writeSnapshots(rejected.id, applicantId);
   564→    } else if (rejected.status !== "Rejected" || rejected.reason !== reason) {
   565→      await db.application.update({ where: { id: rejected.id }, data: { status: "Rejected", reason } });
   566→    }
   567→    const active = rows.find((r) => isActiveStatus(r.status)) ?? null;
   568→    if (!active) {
   569→      const fresh = await db.application.create({
   570→        data: { applicantId, jobId: job.id, status: "Applied", dateApplied: new Date(now - reappliedDaysAgo * day) },
   571→      });
   572→      await writeSnapshots(fresh.id, applicantId);
   573→    } else if (active.status !== "Applied") {
   574→      await db.application.update({ where: { id: active.id }, data: { status: "Applied", reason: null } });
   575→    }
   576→  }
