   173→function TextField({
   174→  label, required, helper, value, onChange, type = "text", placeholder, error, icon, onBlur,
   175→}: {
   176→  label: string; required?: boolean; helper?: string; value: string;
   177→  onChange: (v: string) => void; type?: string; placeholder?: string; error?: string;
   178→  icon?: ReactNode; onBlur?: () => void;
   179→}) {
   180→  return (
   181→    <div>
   182→      {/* Official BEUI Input renders its own label; required fields get a
   183→          trailing " *" (the official label prop is a plain string). Passing
   184→          `error` arms the official shake + red ring + message; every field
   185→          reserves its error line (auth rhythm). `icon` renders the official
   186→          left-icon capsule; onBlur passthrough drives the Task 32 touch. */}
   187→      <Input
   188→        type={type}
   189→        label={required ? `${label} *` : label}
   190→        value={value}
   191→        placeholder={placeholder}
   192→        error={error}
   193→        reserveErrorLine
   194→        leftIcon={icon}
   195→        classNames={{ field: BEUI_FIELD_FILL }}
   196→        onChange={onChange}
   197→        onBlur={onBlur}
   198→      />
   199→      {helper && <p className="mt-1 text-xs text-body-gray">{helper}</p>}
   200→    </div>
   201→  );
   202→}

   204→function SelectField({
   205→  label, required, helper, value, options, onChange,

   207→  label: string; required?: boolean; helper?: string; value: string;
   208→  options: string[]; onChange: (v: string) => void;

   210→  return (

   217→          items must carry plain-string children so the trigger label
   218→          registration resolves (span child = raw value shown). */}
   219→      <Select value={value !== "" ? value : NONE} onValueChange={(v) => onChange(v === NONE ? "" : v)}>
   220→        <SelectTrigger className={inputCls}>
   221→          <span className="sr-only">{label}</span>
   222→          <SelectValue placeholder="Select…" />
   223→        </SelectTrigger>
   224→        <SelectContent>
   225→          {options.map((o) => (
   226→            <SelectItem key={o} value={o}>
   227→              {o}
   228→            </SelectItem>
   229→          ))}
   230→          <SelectItem value={NONE} className="text-stone">
   231→            Not specified
   232→          </SelectItem>
   233→        </SelectContent>
   234→      </Select>
   235→      {helper && <p className="mt-1 text-xs text-body-gray">{helper}</p>}
   236→    </div>
   237→  );

   306→          <Input
   307→            key={f.key}
   308→            type={f.type ?? "text"}
   309→            label={f.label}

   312→            onChange={(next) => onChange({ [f.key]: next })}
   313→          />
   314→        ))}
   315→      </div>
   316→    </div>
   317→  );
   318→}

   322→export default function PersonalInfoSection({
   323→  profile,
   324→  seed,
   325→  refreshSession,
   326→}: {
   327→  profile: ApplicantProfile | null;
   328→  /** Bump to force a re-seed from the server profile (after PDS auto-apply / clear). */
   329→  seed: number;
   330→  refreshSession: () => Promise<unknown>;
   331→}) {
   332→  const [form, setForm] = useState<PersonalForm>(EMPTY_FORM);
   333→  const [refs, setRefs] = useState<CharacterReference[]>([{ ...EMPTY_REF }]);
   334→  const [saveState, setSaveState] = useState<"idle" | "dirty" | "saving" | "saved" | "error">("idle");
   335→  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   336→  const formRef = useRef(form);
   337→  const refsRef = useRef(refs);
   338→  const seededToken = useRef<string>("");
   339→
   340→  // Keep latest-value refs in sync after commit (never during render).
   341→  useEffect(() => {
   342→    formRef.current = form;
   343→    refsRef.current = refs;
   344→  }, [form, refs]);
   345→
   346→  // Seed (and re-seed on `seed` bumps) from the loaded profile — never during
   347→  // silent reloads, so in-flight typing/autosave is never clobbered.
   348→  useEffect(() => {
   349→    if (!profile) return;
   350→    const token = `${seed}`;
   351→    if (seededToken.current === token) return;
   352→    seededToken.current = token;
   353→    setForm(seedForm(profile));
   354→    const list = Array.isArray(profile.characterReferences) ? profile.characterReferences : [];
   355→    setRefs(list.length > 0 ? list.map((r) => ({ ...EMPTY_REF, ...r })) : [{ ...EMPTY_REF }]);
   356→    setSaveState("idle");
   357→  }, [profile, seed]);
   358→
   359→  // Task 32 — official SignUpForm timing: rules are derived on EVERY render
   360→  // (an error clears the moment its field is fixed); `shown` reveals a rule
   361→  // only once the field has been left (onBlur) or Save Changes was pressed.
   362→  // Typed over ALL PersonalForm keys (not just the ones with rules) so the
   363→  // blur touch covers every field in the section.
   364→  const rules: Partial<Record<keyof PersonalForm, string>> = {
   365→    firstName: form.firstName.trim() === "" ? "First Name is required." : undefined,
   366→    lastName: form.lastName.trim() === "" ? "Last Name is required." : undefined,
   367→    emailAddress: form.emailAddress.trim() === "" ? "Email Address is required." : undefined,
   368→    ...(form.adminCase && form.adminCaseDetails.trim() === ""
   369→      ? { adminCaseDetails: "Describe the administrative case to continue." }
   370→      : {}),
   371→  };
   372→  const { shown, touch, touchAll, hasGaps } = useAuthErrors(rules);
   373→
   374→  const save = useCallback(
   375→    async (manual: boolean) => {
   376→      if (timerRef.current) {
   377→        clearTimeout(timerRef.current);
   378→        timerRef.current = null;
   379→      }
   380→      setSaveState("saving");
   381→      try {
   382→        const payload: Record<string, string | boolean | CharacterReference[]> = {
   383→          ...formRef.current,
   384→          characterReferences: refsRef.current
   385→            .map((r) => ({
   386→              name: (r.name ?? "").trim(),
   387→              title: (r.title ?? "").trim(),
   388→              company: (r.company ?? "").trim(),
   389→              companyAddress: (r.companyAddress ?? "").trim(),
   390→              email: (r.email ?? "").trim(),
   391→              contact: (r.contact ?? "").trim(),
   392→            }))
   393→            .filter((r) => r.name !== ""), // empty-name rows filtered at save (§7.4)
   394→        };
   395→        await apiFetch("/api/applicant/profile", { method: "PUT", body: payload });
   396→        setSaveState("saved");
   397→        if (manual) {
   398→          toast.success("Profile changes saved");
   399→          await refreshSession();
   400→        }
   401→      } catch (e) {
   402→        setSaveState("error");
   403→        if (manual) toast.error(e instanceof Error ? e.message : "Save failed. Please try again");
   404→      }
   405→    },
   406→    [refreshSession]
   407→  );
   408→
   409→  /** Re-arm the 1.2 s autosave debounce on every change (§7.4). Never
   410→   *  triggered by the Task 32 error touch — only real edits save. */
   411→  const armAutosave = useCallback(() => {
   412→    setSaveState("dirty");
   413→    if (timerRef.current) clearTimeout(timerRef.current);
   414→    timerRef.current = setTimeout(() => {
   415→      timerRef.current = null;
   416→      void save(false);
   417→    }, 1200);
   418→  }, [save]);
   419→
   420→  useEffect(
   421→    () => () => {
   422→      if (timerRef.current) clearTimeout(timerRef.current);
   423→    },
   424→    []
   425→  );
   426→
   427→  const set = useCallback(
   428→    <K extends keyof PersonalForm>(key: K, value: PersonalForm[K]) => {
   429→      setForm((prev) => ({ ...prev, [key]: value }));
   430→      armAutosave();
   431→    },
   432→    [armAutosave]
   433→  );
   434→
   435→  /** Manual save — Task 32 official SignUpForm timing: Save Changes reveals
   436→   *  every gap at once (touchAll) and the save only proceeds when the live
   437→   *  rules are clean; correcting a field clears its rule instantly. */
   438→  const saveManual = useCallback(() => {
   439→    touchAll();
   440→    if (hasGaps) return;
   441→    void save(true);
   442→  }, [touchAll, hasGaps, save]);
   443→
   444→  const identityDone = filledCount(form, IDENTITY_FIELDS);
   445→  const addressDone = filledCount(form, ADDRESS_FIELDS);
   446→  const legalTotal = 2 + (form.adminCase ? 1 : 0) + (form.crimeCharge ? 2 : 0);
   447→  const legalDone =
   448→    2 +
   449→    (form.adminCase && form.adminCaseDetails.trim() !== "" ? 1 : 0) +
   450→    (form.crimeCharge && form.crimeDate.trim() !== "" ? 1 : 0) +
   451→    (form.crimeCharge && form.crimeCaseStatus.trim() !== "" ? 1 : 0);
   452→  const refsNamed = refs.filter((r) => (r.name ?? "").trim() !== "").length;
   453→
   454→  return (
   455→    <SectionCard
   456→      variant="plain"
   457→      bareOnMobile
   458→      icon={UserRound}
   459→      chipTone="slate"
   460→      title="Personal Information"
   461→      description="Your identity, contact details, and legal declarations."
   462→      actions={
   463→        <div className="flex items-center gap-3">
   464→          <SaveHint
   465→            saving={saveState === "saving"}
   466→            saved={saveState === "saved"}
   467→            dirty={saveState === "dirty"}
   468→            error={saveState === "error" ? "Save failed. Click Save Changes" : undefined}
   469→          />
   470→          <StatefulButton
   471→            type="button"
   472→            onClick={saveManual}
   473→            state={saveState === "saving" ? "loading" : "idle"}
   474→            loadingText="Saving"
   475→            className="min-h-[44px] px-5 py-2.5 text-sm"
   476→          >
   477→            Save Changes
   478→          </StatefulButton>
   479→        </div>
   480→      }
   481→    >
   482→      <div className="space-y-3">
   483→        {/* Group 1 — Identity */}
   484→        <CollapsibleGroup icon={UserRound} title="Identity" done={identityDone} total={IDENTITY_FIELDS.length}>
   485→          <div className="grid grid-cols-1 gap-x-3 gap-y-1 sm:grid-cols-2">
   486→            <TextField label="First Name" required placeholder="e.g. Juan" icon={<User />} value={form.firstName} error={shown("firstName")} onChange={(v) => set("firstName", v)} onBlur={() => touch("firstName")} />
   487→            <TextField label="Middle Name" icon={<User />} value={form.middleName} onChange={(v) => set("middleName", v)} onBlur={() => touch("middleName")} />
   488→            <TextField label="Last Name" required placeholder="e.g. Dela Cruz" icon={<User />} value={form.lastName} error={shown("lastName")} onChange={(v) => set("lastName", v)} onBlur={() => touch("lastName")} />
   489→            <TextField
   490→              label="Extension Name"
   491→              helper="e.g. Jr., Sr., III"
   492→              icon={<User />}
   493→              value={form.extensionName}
   494→              onChange={(v) => set("extensionName", v)}
   495→              onBlur={() => touch("extensionName")}
   496→            />
   497→            <TextField
   498→              label="Email Address"
   499→              required
   500→              type="email"
   501→              helper="Interview invites are sent to this address"
   502→              icon={<Mail />}
   503→              value={form.emailAddress}
   504→              error={shown("emailAddress")}
   505→              onChange={(v) => set("emailAddress", v)}
   506→              onBlur={() => touch("emailAddress")}
   507→            />
   508→            <TextField
   509→              label="Mobile Number"
   510→              type="tel"
   511→              helper="09XXXXXXXXX"
   512→              icon={<Phone />}
   513→              value={form.mobileNumber}
   514→              onChange={(v) => set("mobileNumber", v)}
   515→              onBlur={() => touch("mobileNumber")}
   516→            />
