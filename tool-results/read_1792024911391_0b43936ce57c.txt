   254→/** Plain-text posting section rendered as a disc bullet list. */
   255→function TextBullets({ text }: { text: string }) {
   256→  const items = text
   257→    .split(/\r?\n/)
   258→    .map((l) => l.replace(/^[-•*]\s*/, "").trim())
   259→    .filter(Boolean);
   260→  if (items.length === 0) return null;
   261→  return (
   262→    <ul className="list-disc space-y-2 pl-5 text-base leading-relaxed text-stone">
   263→      {items.map((item, i) => (
   264→        <li key={i}>{item}</li>
   265→      ))}
   266→    </ul>
   267→  );
   268→}

   270→/** Posting section: sanitized rich HTML when present, else text bullets. */
   271→function PostingSection({ title, html, text }: { title: string; html: string | null; text: string | null }) {
   272→  if (html && html.trim()) {
   273→    return (
   274→      <section className="space-y-3">
   275→        <h3 className="text-lg font-medium text-ink">{title}</h3>
   276→        <div className="rich-text text-base leading-relaxed text-stone" dangerouslySetInnerHTML={{ __html: html }} />
   277→      </section>
   278→    );
   279→  }
   280→  if (text && text.trim()) {
   281→    return (
   282→      <section className="space-y-3">
   283→        <h3 className="text-lg font-medium text-ink">{title}</h3>
   284→        <TextBullets text={text} />
   285→      </section>
   286→    );
   287→  }
   288→  return null;
   289→}

   424→export default function PublicLanding() {
   425→  const [jobs, setJobs] = useState<JobWire[] | null>(null);
   426→  const [error, setError] = useState<string | null>(null);
   427→  // Last non-null error — keeps the BEUI alert's exit animation populated with
   428→  // content while the message state has already cleared.
   429→  const [lastError, setLastError] = useState<string | null>(null);
   430→  useEffect(() => {
   431→    if (error !== null) setLastError(error);
   432→  }, [error]);
   433→  const [selectedId, setSelectedId] = useState<number | null>(null);
   434→  const [expandedId, setExpandedId] = useState<number | null>(null);
   435→  // Single-line browse filter (user direction: the hero CTA and the header
   436→  // board link are gone — the frontpage register is filtered by a
   437→  // [search bar] + [Sort] row, the board's mobile Job Openings pattern).
   438→  const [query, setQuery] = useState("");
   439→  const [sort, setSort] = useState<SortKey>("newest");
   440→  const [isDesktop, setIsDesktop] = useState(false);
   441→  const lastJson = useRef("");
   442→  const paneRef = useRef<HTMLDivElement>(null);
   443→  const { saved, toggle: toggleSave } = useSavedJobs();
   444→  // Shared apply gate — anonymous visitors get the sign-in prompt, complete
   445→  // applicants get the confirm flow (same as the board), incomplete profiles
   446→  // are sent to finish their profile first (the fast-track was retired).
   447→  const { requestApply, dialogs } = useApplyFlow();
   448→
   449→  const load = useCallback(async (silent: boolean) => {
   450→    try {
   451→      const data = await apiFetch<JobWire[]>("/api/jobs?limit=200");
   452→      const json = JSON.stringify(data);
   453→      if (!silent || json !== lastJson.current) {
   454→        lastJson.current = json;
   455→        setJobs(data);
   456→      }
   457→      setError(null);
   458→    } catch (e) {
   459→      if (!silent) setError(e instanceof Error ? e.message : "Failed to load positions");
   460→    }
   461→  }, []);
   462→
   463→  // Mount + 30s silent poll + refetch on focus (spec §13 cadence).
   464→  useEffect(() => {
   465→    void load(false);
   466→    const timer = setInterval(() => {
   467→      if (!document.hidden) void load(true);
   468→    }, 30_000);
   469→    const onFocus = () => void load(true);
   470→    window.addEventListener("focus", onFocus);
   471→    return () => {
   472→      clearInterval(timer);
   473→      window.removeEventListener("focus", onFocus);
   474→    };
   475→  }, [load]);
   476→
   477→  // Breakpoint sync — desktop keeps both panes; mobile swaps list ↔ detail.
   478→  useEffect(() => {
   479→    const mq = window.matchMedia("(min-width: 1024px)");
   480→    const sync = () => setIsDesktop(mq.matches);
   481→    sync();
   482→    mq.addEventListener("change", sync);
   483→    return () => mq.removeEventListener("change", sync);
   484→  }, []);
   485→
   486→  const openJobs = useMemo(
   487→    () => (jobs ?? []).filter((j) => !deadlineState(j.deadlineDate).overdue),
   488→    [jobs]
   489→  );
   490→
   491→  // Filter + sort over the open corpus — the needle covers every line the
   492→  // register cards actually display: title, position title, item number,
   493→  // division name AND place of assignment (a visitor typing "metrology"
   494→  // must match the card that reads "Metrology Laboratory"). Sort uses the
   495→  // same three comparators as the board (Newest · Deadline · Salary desc).
   496→  const filteredOpenJobs = useMemo(() => {
   497→    let list = openJobs;
   498→    const needle = query.trim().toLowerCase();
   499→    if (needle) {
   500→      list = list.filter((j) =>
   501→        `${j.title} ${j.position?.positionTitle ?? ""} ${j.position?.itemNumber ?? ""} ${
   502→          j.position?.division ? divisionFullName(j.position.division) : ""
   503→        } ${jobPlace(j)}`
   504→          .toLowerCase()
   505→          .includes(needle)
   506→      );
   507→    }
   508→    const deadlineTime = (j: JobWire) =>
   509→      j.deadlineDate ? new Date(j.deadlineDate).getTime() : Number.POSITIVE_INFINITY;
   510→    const sorted = [...list];
   511→    if (sort === "newest") {
   512→      sorted.sort((a, b) => Date.parse(b.publishedAt) - Date.parse(a.publishedAt));
   513→    } else if (sort === "deadline") {
   514→      sorted.sort((a, b) => deadlineTime(a) - deadlineTime(b));
   515→    } else {
   516→      sorted.sort((a, b) => (b.position?.salaryAmount ?? -1) - (a.position?.salaryAmount ?? -1));
   517→    }
   518→    return sorted;
   519→  }, [openJobs, query, sort]);
   520→
   521→  const selectedJob = useMemo(
   522→    () => openJobs.find((j) => j.id === selectedId) ?? null,
   523→    [openJobs, selectedId]
   524→  );
   525→
   526→  // Desktop opens with the first posting pre-selected (reference-board
   527→  // behavior) and snaps to the top of the FILTERED register whenever the
   528→  // current selection is filtered out (search/sort) or closed by a poll —
   529→  // the posting pane never lags the register.
   530→  useEffect(() => {
   531→    if (
   532→      isDesktop &&
   533→      filteredOpenJobs.length > 0 &&
   534→      !filteredOpenJobs.some((j) => j.id === selectedId)
   535→    ) {
   536→      setSelectedId(filteredOpenJobs[0].id);
   537→    }
   538→  }, [isDesktop, filteredOpenJobs, selectedId]);
   539→
   540→  useEffect(() => {
   541→    if (selectedId !== null && !openJobs.some((j) => j.id === selectedId)) {
   542→      setSelectedId(openJobs[0]?.id ?? null);
   543→    }
   544→  }, [openJobs, selectedId]);
   545→
   546→  // Reference-board behavior — a freshly picked posting opens from the top,
   547→  // even when the previous posting was scrolled deep into its body.
   548→  useEffect(() => {
   549→    if (paneRef.current) paneRef.current.scrollTop = 0;
   550→  }, [selectedId]);
   551→
   552→  const select = useCallback(
   553→    (id: number) => {
   554→      setSelectedId(id);
   555→      if (!window.matchMedia("(min-width: 1024px)").matches) {
   556→        window.scrollTo({ top: 0, behavior: "smooth" });
   557→      }
   558→    },
   559→    []
   560→  );
   561→
   562→  const saveAndToast = useCallback(
   563→    (job: JobWire) => {
   564→      const wasSaved = saved.includes(job.id);
   565→      toggleSave(job.id);
   566→      toast.success(
   567→        wasSaved
   568→          ? "Removed from your saved jobs"
   569→          : `Saved ${humanize(job.position?.positionTitle || job.title)}`
   570→      );
   571→    },
   572→    [saved, toggleSave]
   573→  );
   574→
   575→  const showList = isDesktop || selectedId === null;
   576→  const showDetail = isDesktop || selectedId !== null;
   577→
   578→  return (
   579→    // Viewport-adaptive board (no app pinning): a normal min-h-screen flex
   580→    // column whose footer always sits at the real bottom — pinned to the
   581→    // viewport on short pages (mt-auto), pushed down naturally on tall ones.
   582→    // Only the posting pane is sticky (viewport-capped); the register is a
   583→    // plain block that scrolls with the page — see the grid below.
   584→    <div className="flex min-h-screen flex-col">
   585→      <SiteHeader />
   586→
   587→      {/* Wide-screen grid (QA: oversized side margins at 1200px on wide
   588→          monitors) — the landing now shares the workspace shell's 1600px
   589→          page max-width; the floating header pill matches. */}
   590→      <main className="mx-auto w-full max-w-[1600px] flex-1 px-4 pb-10 pt-6 sm:px-6 lg:px-8">
   591→        {/* Hero — "Hero Sky" wash field (rounded-3xl = 24px editorial imagery step) anchoring the explorer:
   592→            the Jeton heading step (text-display, 44px) keeps the headline
   593→            institutional rather than poster-scale, then the agency name as
   594→            the quiet proof strip (QA: the emblem itself was dropped — the
   595→            sticky header already carries the DOST mark, and a second one 40px
   596→            below read as a duplicate), then the single .dlg-cta RMIS Electric
   597→            Blue capsule CTA (real marks only). */}
   598→        <section className="hero-sky rounded-3xl p-6 sm:p-8 lg:px-10 lg:py-10">
   599→          {/* QA 2026-02: header renamed to "Job Opportunities" to match the
   600→              Positions tab; the live "N open" pill was removed — it read as
   601→              an unclickable button. */}
   602→          <h1 className="text-display font-display">Job Opportunities</h1>
   603→          {/* Institutional proof — agency name only; the emblem lives in the
   604→              sticky header directly above, so repeating it here read as a
   605→              duplicate seal (user QA). */}
   606→          <div className="mt-4">
   607→            <p className="min-w-0 text-sm font-medium leading-snug text-ink">
   608→              Metals Industry Research and Development Center
   609→            </p>
   610→          </div>
   611→        </section>
   612→
   613→        <Alert variant="destructive" open={error !== null} className="mt-6">
   614→          <AlertIcon />
   615→          <AlertContent>
   616→            <AlertDescription>{error ?? lastError}</AlertDescription>
   617→            {error !== null && (
   618→              <AlertAction>
   619→                <button
   620→                  onClick={() => void load(false)}
   621→                  className="inline-flex min-h-[44px] items-center text-sm font-medium text-metric-blue underline underline-offset-2"
   622→                >
   623→                  Try again
   624→                </button>
   625→              </AlertAction>
   626→            )}
   627→          </AlertContent>
   628→        </Alert>
   629→        {error !== null ? null : jobs === null ? (
   630→          <div className="mt-6 space-y-3.5" aria-hidden="true">
   631→            {[0, 1, 2].map((i) => (
   632→              <div key={i} className="skel h-28" />
   633→            ))}
   634→          </div>
   635→        ) : openJobs.length === 0 ? (
   636→          <div className="dlg-card mt-6">
   637→            <EmptyState
   638→              icon={Inbox}
   639→              title="No open positions right now"
   640→              description="Check back soon or explore the full board for upcoming announcements."
   641→            />
   642→            <div className="pb-8 text-center">
   643→              <button
   644→                onClick={() => navigate("vacant-positions")}
   645→                className="inline-flex min-h-[44px] items-center gap-2 text-sm text-metric-blue underline underline-offset-2 hover:text-ink"
   646→              >
   647→                Go to the full positions board
   648→                <ArrowRight className="h-4 w-4" aria-hidden="true" />
   649→              </button>
   650→            </div>
   651→          </div>
   652→        ) : (
   653→          // Viewport-adaptive two-pane board. The posting pane sizes to its
   654→          // own content, capped at the LIVE viewport height (100dvh − 88px
   655→          // sticky offset − 24px breathing room — no fixed pixel constant):
   656→          //   • tall screens — the complete description renders in full with
   657→          //     no internal scrolling; the footer only ever sits BELOW it;
   658→          //   • short screens — the pane caps at the available space and
   659→          //     scrolls internally while pinned under the header, and the
   660→          //     grid-row clamp keeps it above the footer at every position.
   661→          // Sticky needs TRAVEL ROOM: a sticky element only stays put while
   662→          // its containing row is TALLER than itself, and the row height is
   663→          // max(register, posting). When the register is shorter than the
   664→          // posting (few vacancies — e.g. the sandbox's two cards), the row
   665→          // would equal the pane's height and the pane could never pin — it
   666→          // would just scroll away with the page. So the row reserves at
   667→          // least 100dvh + 128px (= the pane's cap + a 240px pin phase on
   668→          // cap-limited screens): the pane rides up with the hero, pins at
   669→          // top-88 for ~240px of scroll, then yields only as the footer
   670→          // rises. With a tall register the row is the register's natural
   671→          // height and this floor is inert.
   672→          // The job list is deliberately NOT sticky — a plain block that
   673→          // scrolls with the page (user clarification).
   674→          <div className="mt-6 grid gap-8 sm:mt-8 lg:min-h-[calc(100dvh+128px)] lg:grid-cols-[minmax(340px,430px)_minmax(0,1fr)]">
   675→            {/* Left · the register — plain block, natural height, scrolls
   676→                with the page (NOT sticky, per user clarification). */}
   677→            {showList && (
   678→              <div
   679→                aria-label="Job opportunities list"
   680→                className="scroll-thin"
   681→              >
   682→                {/* Single-line browse filter — [search bar] + [Sort] on one
   683→                    row above the register (user direction: the retired hero
   684→                    CTA / header board link are replaced by this). Sort
   685→                    options match the desktop board's trio. */}
   686→                <div className="flex items-center gap-2">
   687→                  <Input
   688→                    id="landing-search"
   689→                    aria-label="Search positions"
   690→                    leftIcon={<Search />}
   691→                    value={query}
   692→                    onChange={setQuery}
   693→                    placeholder="Search positions…"
   694→                    className="min-w-0 flex-1"
   695→                  />
   696→                  <MorphSelect
   697→                    className="w-[124px] shrink-0"
   698→                    value={sort}
   699→                    onValueChange={(v) => setSort(v as SortKey)}
   700→                  >
   701→                    <MorphSelectTrigger className="min-h-[44px]">
   702→                      <span className="sr-only">Sort job openings</span>
   703→                      <MorphSelectValue />
   704→                    </MorphSelectTrigger>
   705→                    <MorphSelectContent>
   706→                      <MorphSelectItem value="newest">Newest</MorphSelectItem>
   707→                      <MorphSelectItem value="deadline">Deadline</MorphSelectItem>
   708→                      <MorphSelectItem value="salary">Salary desc</MorphSelectItem>
   709→                    </MorphSelectContent>
   710→                  </MorphSelect>
   711→                </div>
   712→
   713→                {filteredOpenJobs.length === 0 ? (
   714→                  <div className="dlg-card mt-3.5">
   715→                    <EmptyState
   716→                      icon={Search}
   717→                      title="No positions match your search"
   718→                      description="Try a different term or clear the search to see every opening."
   719→                      action={
   720→                        <button
   721→                          onClick={() => setQuery("")}
   722→                          className="dlg-ghost min-h-[44px] px-6 text-sm"
   723→                        >
   724→                          Clear search
   725→                        </button>
   726→                      }
   727→                    />
   728→                  </div>
   729→                ) : (
   730→                  <div className="mt-3.5 space-y-3.5">
   731→                    {filteredOpenJobs.map((job) => (
   732→                      <JobListCard
   733→                        key={job.id}
   734→                        job={job}
   735→                        selected={selectedId === job.id}
   736→                        expanded={expandedId === job.id}
   737→                        saved={saved.includes(job.id)}
   738→                        onSelect={() => select(job.id)}
   739→                        onToggle={() => setExpandedId(expandedId === job.id ? null : job.id)}
   740→                        onSave={() => saveAndToast(job)}
   741→                      />
   742→                    ))}
   743→                  </div>
   744→                )}
   745→              </div>
   746→            )}
   747→
   748→            {/* Right · the posting — ONE sticky, viewport-capped scroller
   749→                (paneRef sits on the scroller itself, so the select-reset
   750→                actually zeroes it). Its natural height is the full posting,
   751→                capped at the live viewport: complete content with no inner
   752→                scroll when the screen has room; capped + internally
   753→                scrollable when it doesn't. Sticky keeps it in view while the
   754→                list scrolls with the page; the grid-row clamp keeps it above
   755→                the footer. On mobile it is a plain block (single-pane swap,
   756→                page scroll). */}
   757→            {showDetail && (
   758→              <div
   759→                ref={paneRef}
   760→                className="scroll-thin lg:sticky lg:top-[88px] lg:self-start lg:max-h-[calc(100dvh-112px)] lg:overflow-y-auto lg:pr-2"
   761→              >
   762→                {selectedJob ? (
   763→                  <div className="dlg-card p-6 sm:p-8">
   764→                    <JobDetail
   765→                      job={selectedJob}
   766→                      saved={saved.includes(selectedJob.id)}
   767→                      onApply={() => requestApply(selectedJob)}
   768→                      onSave={() => saveAndToast(selectedJob)}
   769→                      onBack={() => setSelectedId(null)}
   770→                    />
   771→                  </div>
   772→                ) : (
   773→                  <div className="dlg-card">
   774→                    <EmptyState
   775→                      icon={Briefcase}
   776→                      title="Select a position"
   777→                      description="Choose a role from the list to read the full posting."
   778→                    />
   779→                  </div>
   780→                )}
   781→              </div>
   782→            )}
   783→          </div>
   784→        )}
   785→
   786→      </main>
   787→
   788→      <Footer wide />
   789→      {/* Shared apply dialog (confirm / MQR / done), mounted once. */}
   790→      {dialogs}
   791→    </div>
   792→  );
   793→}
