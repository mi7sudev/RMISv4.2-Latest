     4→// RMIS — Public landing (spec §7.1): a two-pane job explorer in the
     5→// reference-board style — TOP: a Bevel "Hero Sky" rounded-32 gradient field
     6→// in a split composition (Google-Workspace rhythm): display headline + copy
     7→// + .dlg-cta RMIS Electric Blue capsule CTA on the left, the official DOST-MIRDC brand mark
     8→// with the agency authority line as the institutional proof block on the
     9→// right; LEFT: a register of Cloud Card
    10→// job cards (title · company · place · salary · clamped snippet · posted
    11→// age · chevron quick-view + bookmark); RIGHT: the full posting pane
    12→// (display title, verified company row + "View all jobs", icon meta rows,
    13→// posted age, brand-blue Quick apply + ghost Save, description,
    14→// Responsibilities / Qualifications / Compensation sections). The selected
    15→// card carries the brand-blue ring (RMIS Electric Blue #1591dc). Only the
    16→// posting pane is sticky: it grows to its full content height when the
    17→// screen has room (complete description, no internal scrolling) and caps at
    18→// the live viewport — scrolling internally while pinned under the header —
    19→// when it doesn't, never cut off by or hidden behind the footer. The board
    20→// row always reserves at least a full viewport plus a 240px pin phase
    21→// (min-h 100dvh+128px), so the pane has travel room to stay pinned while
    22→// the page scrolls even when the register is shorter than the posting. The
    23→// job list is a plain block that scrolls with the page. Mobile swaps panes
    24→// (list → detail with back).
    25→// Data from GET /api/jobs (30s silent poll + refetch on focus); visitors
    26→// without a session are routed to sign-in by the shared apply gate; saves
    27→// persist in localStorage. Apply actions are cancelled-only on the landing.

   239→/** Plain-text posting section rendered as a disc bullet list. */
   240→function TextBullets({ text }: { text: string }) {
   241→  const items = text
   242→    .split(/\r?\n/)
   243→    .map((l) => l.replace(/^[-•*]\s*/, "").trim())
   244→    .filter(Boolean);
   245→  if (items.length === 0) return null;
   246→  return (
   247→    <ul className="list-disc space-y-2 pl-5 text-base leading-relaxed text-stone">
   248→      {items.map((item, i) => (
   249→        <li key={i}>{item}</li>
   250→      ))}
   251→    </ul>
   252→  );
   253→}
   254→
   255→/** Posting section: sanitized rich HTML when present, else text bullets. */
   256→function PostingSection({ title, html, text }: { title: string; html: string | null; text: string | null }) {
   257→  if (html && html.trim()) {
   258→    return (
   259→      <section className="space-y-3">
   260→        <h3 className="text-lg font-medium text-ink">{title}</h3>
   261→        <div className="rich-text text-base leading-relaxed text-stone" dangerouslySetInnerHTML={{ __html: html }} />
   262→      </section>
   263→    );
   264→  }
   265→  if (text && text.trim()) {
   266→    return (
   267→      <section className="space-y-3">
   268→        <h3 className="text-lg font-medium text-ink">{title}</h3>
   269→        <TextBullets text={text} />
   270→      </section>
   271→    );
   272→  }
   273→  return null;
   274→}

   276→function JobDetail({
   277→  job,
   278→  saved,
   279→  onApply,
   280→  onSave,
   281→  onBack,
   282→}: {
   283→  job: JobWire;
   284→  saved: boolean;
   285→  onApply: () => void;
   286→  onSave: () => void;
   287→  onBack: () => void;
   288→}) {
   289→  const pos = job.position;
   290→  const title = pos?.positionTitle || job.title || "Position Title Unavailable";
   291→  const salary = pos?.salaryAmount ?? null;
   292→  const deadline = job.deadlineDate;
   293→  const overdue = jobDeadlinePassed(deadline);
   294→
   295→  const meta: { icon: typeof MapPin; label: string; num?: boolean; danger?: boolean }[] = [];
   296→  if (job.position?.division) meta.push({ icon: Building2, label: job.position.division });
   297→  if (job.positionType) meta.push({ icon: Briefcase, label: humanize(job.positionType) });
   298→  if (salary != null)
   299→    meta.push({
   300→      icon: Banknote,
   301→      label: `${formatCurrency(salary)} per month${
   302→        pos?.salaryGrade ? ` · SG ${pos.salaryGrade}${pos.salaryStep ? `/${pos.salaryStep}` : ""}` : ""
   303→      }`,
   304→      num: true,
   305→    });
   306→  meta.push({
   307→    icon: Clock,
   308→    label: overdue ? "Closed: deadline passed" : deadline ? `Closes ${formatDate(deadline)}` : "Open until filled",
   309→    num: !overdue && !deadline,
   310→    danger: overdue,
   311→  });
   312→
   313→  return (
   314→    <div>
   315→      {/* Mobile — back to the register */}
   316→      <button
   317→        type="button"
   318→        onClick={onBack}
   319→        className="focus-ring mb-4 inline-flex items-center gap-1.5 text-sm font-medium text-metric-blue lg:hidden"
   320→      >
   321→        <ChevronLeft className="h-4 w-4" aria-hidden="true" /> Back to job list
   322→      </button>
   323→
   324→      {/* Posting title — the house page-title step (24px). The pane is a
   325→          document view: it must sit below the 44px hero, not compete. */}
   326→      <h2 className="text-2xl font-medium leading-tight text-ink">
   327→        {humanize(title)}
   328→      </h2>
   329→      <div className="mt-2.5 flex flex-wrap items-center gap-x-2.5 gap-y-1.5">
   330→        <span className="text-base font-medium text-ink">DOST-MIRDC</span>
   331→        <BadgeCheck className="h-[18px] w-[18px] shrink-0 text-stone" aria-label="Verified agency" />
   332→        <button
   333→          type="button"
   334→          onClick={() => navigate("vacant-positions")}
   335→          className="focus-ring text-[15px] text-metric-blue underline underline-offset-2 transition-colors hover:text-ink"
   336→        >
   337→          View all jobs
   338→        </button>
   339→      </div>
   340→
   341→      <ul className="mt-4 space-y-2.5">
   342→        <li className="flex items-start gap-2.5 text-base text-stone">
   343→          <MapPin className="mt-0.5 h-[18px] w-[18px] shrink-0 text-stone" aria-hidden="true" />
   344→          {jobPlace(job)}
   345→        </li>
   346→        {meta.map((m, i) => (
   347→          <li
   348→            key={i}
   349→            className={cn(
   350→              "flex items-start gap-2.5 text-base",
   351→              m.danger ? "font-medium text-[var(--bad)]" : "text-stone"
   352→            )}
   353→          >
   354→            <m.icon className="mt-0.5 h-[18px] w-[18px] shrink-0 text-stone" aria-hidden="true" />
   355→            <span className={cn(m.num && "num")}>{m.label}</span>
   356→          </li>
   357→        ))}
   358→      </ul>
   359→
   360→      <p className="num mt-4 text-sm text-stone">Posted {timeAgo(postedAt(job))}</p>
   361→
   362→      <div className="mt-5 flex flex-wrap items-center gap-3">
   363→        <button type="button" onClick={onApply} disabled={overdue} className={ctaBtn}>
   364→          Quick apply
   365→        </button>
   366→        <button
   367→          type="button"
   368→          onClick={onSave}
   369→          aria-pressed={saved}
   370→          className={ghostBtn}
   371→        >
   372→          <Bookmark className={cn("h-4 w-4", saved && "fill-current")} aria-hidden="true" />
   373→          {saved ? "Saved" : "Save"}
   374→        </button>
   375→        {overdue && (
   376→          <span className="flex items-center gap-1.5 text-xs font-medium text-[var(--bad)]">
   377→            <AlertCircle className="h-3.5 w-3.5" aria-hidden="true" /> Deadline passed
   378→          </span>
   379→        )}
   380→      </div>
   381→
   382→      {/* Posting body */}
   383→      <div className="mt-8 space-y-8 border-t border-divider pt-8">
   384→        {job.briefDescription &&
   385→          job.briefDescription
   386→            .split(/\n{2,}/)
   387→            .filter((p) => p.trim())
   388→            .map((p, i) => (
   389→              <p key={i} className="text-base leading-relaxed text-stone">
   390→                {p}
   391→              </p>
   392→            ))}
   393→        <PostingSection title="Responsibilities" html={job.dutiesHtml} text={job.dutiesResponsibilities} />
   394→        <PostingSection
   395→          title="Qualifications"
   396→          html={job.otherQualificationsHtml}
   397→          text={job.otherQualifications}
   398→        />
   399→        <PostingSection title="Compensation" html={job.compensationHtml} text={job.compensationPackage} />
   400→      </div>
   401→    </div>
   402→  );
   403→}

   407→export default function PublicLanding() {
   408→  const [jobs, setJobs] = useState<JobWire[] | null>(null);
   409→  const [error, setError] = useState<string | null>(null);
   410→  // Last non-null error — keeps the BEUI alert's exit animation populated with
   411→  // content while the message state has already cleared.
   412→  const [lastError, setLastError] = useState<string | null>(null);
   413→  useEffect(() => {
   414→    if (error !== null) setLastError(error);
   415→  }, [error]);
   416→  const [selectedId, setSelectedId] = useState<number | null>(null);
   417→  const [expandedId, setExpandedId] = useState<number | null>(null);
   418→  const [isDesktop, setIsDesktop] = useState(false);
   419→  const lastJson = useRef("");
   420→  const paneRef = useRef<HTMLDivElement>(null);
   421→  const { saved, toggle: toggleSave } = useSavedJobs();
   422→  // Shared apply gate — anonymous visitors get the sign-in prompt, complete
   423→  // applicants get the confirm flow (same as the board), incomplete profiles
   424→  // are sent to finish their profile first (the fast-track was retired).
   425→  const { requestApply, dialogs } = useApplyFlow();
   426→
   427→  const load = useCallback(async (silent: boolean) => {
   428→    try {
   429→      const data = await apiFetch<JobWire[]>("/api/jobs?limit=200");
   430→      const json = JSON.stringify(data);
   431→      if (!silent || json !== lastJson.current) {
   432→        lastJson.current = json;
   433→        setJobs(data);
   434→      }
   435→      setError(null);
   436→    } catch (e) {
   437→      if (!silent) setError(e instanceof Error ? e.message : "Failed to load positions");
   438→    }
   439→  }, []);
   440→
   441→  // Mount + 30s silent poll + refetch on focus (spec §13 cadence).
   442→  useEffect(() => {
   443→    void load(false);
   444→    const timer = setInterval(() => {
   445→      if (!document.hidden) void load(true);
   446→    }, 30_000);
   447→    const onFocus = () => void load(true);
   448→    window.addEventListener("focus", onFocus);
   449→    return () => {
   450→      clearInterval(timer);
   451→      window.removeEventListener("focus", onFocus);
   452→    };
   453→  }, [load]);
   454→
   455→  // Breakpoint sync — desktop keeps both panes; mobile swaps list ↔ detail.
   456→  useEffect(() => {
   457→    const mq = window.matchMedia("(min-width: 1024px)");
   458→    const sync = () => setIsDesktop(mq.matches);
   459→    sync();
   460→    mq.addEventListener("change", sync);
   461→    return () => mq.removeEventListener("change", sync);
   462→  }, []);
   463→
   464→  const openJobs = useMemo(
   465→    () => (jobs ?? []).filter((j) => !deadlineState(j.deadlineDate).overdue),
   466→    [jobs]
   467→  );
   468→
   469→  const selectedJob = useMemo(
   470→    () => openJobs.find((j) => j.id === selectedId) ?? null,
   471→    [openJobs, selectedId]
   472→  );
   473→
   474→  // Desktop opens with the first posting pre-selected (reference-board
   475→  // behavior); a stale selection after a poll snaps back to the top card.
   476→  useEffect(() => {
   477→    if (isDesktop && selectedId === null && openJobs.length > 0) {
   478→      setSelectedId(openJobs[0].id);
   479→    }
   480→  }, [isDesktop, openJobs, selectedId]);
   481→
   482→  useEffect(() => {
   483→    if (selectedId !== null && !openJobs.some((j) => j.id === selectedId)) {
   484→      setSelectedId(openJobs[0]?.id ?? null);
   485→    }
   486→  }, [openJobs, selectedId]);
   487→
   488→  // Reference-board behavior — a freshly picked posting opens from the top,
   489→  // even when the previous posting was scrolled deep into its body.
   490→  useEffect(() => {
   491→    if (paneRef.current) paneRef.current.scrollTop = 0;
   492→  }, [selectedId]);
   493→
   494→  const select = useCallback(
   495→    (id: number) => {
   496→      setSelectedId(id);
   497→      if (!window.matchMedia("(min-width: 1024px)").matches) {
   498→        window.scrollTo({ top: 0, behavior: "smooth" });
   499→      }
   500→    },
   501→    []
   502→  );
   503→
   504→  const saveAndToast = useCallback(
   505→    (job: JobWire) => {
   506→      const wasSaved = saved.includes(job.id);
   507→      toggleSave(job.id);
   508→      toast.success(
   509→        wasSaved
   510→          ? "Removed from your saved jobs"
   511→          : `Saved ${humanize(job.position?.positionTitle || job.title)}`
   512→      );
   513→    },
   514→    [saved, toggleSave]
   515→  );
   516→
   517→  const showList = isDesktop || selectedId === null;
   518→  const showDetail = isDesktop || selectedId !== null;
   519→
   520→  return (
   521→    // Viewport-adaptive board (no app pinning): a normal min-h-screen flex
   522→    // column whose footer always sits at the real bottom — pinned to the
   523→    // viewport on short pages (mt-auto), pushed down naturally on tall ones.
   524→    // Only the posting pane is sticky (viewport-capped); the register is a
   525→    // plain block that scrolls with the page — see the grid below.
