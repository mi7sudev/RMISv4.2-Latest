   173→   173→/** ISO date (YYYY-MM-DD, or a full ISO datetime) → "May 12, 1990".
   174→   174→ *  Unparseable input falls back to the raw string. */
   175→   175→function formatDate(v: string | null | undefined): string {
   176→   176→  const raw = (v ?? "").trim();
   177→   177→  if (raw === "") return "";
   178→   178→  const iso = raw.length >= 10 ? raw.slice(0, 10) : raw;
   179→   179→  const d = new Date(`${iso}T00:00:00`);
   180→   180→  if (Number.isNaN(d.getTime())) return raw;
   181→   181→  return d.toLocaleDateString("en-PH", { year: "numeric", month: "long", day: "numeric" });
   182→   182→}
