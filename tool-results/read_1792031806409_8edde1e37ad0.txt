    99→export async function sendEmail(input: EmailInput): Promise<SendResult> {
   100→  let result: SendResult;
   101→  try {
   102→    if (!input.to || !input.to.includes("@")) {
   103→      result = { status: "skipped", error: "No valid recipient address", provider: EMAIL_PROVIDER };
   104→    } else if (EMAIL_PROVIDER === "middleware" && EMAIL_MIDDLEWARE_URL) {
   105→      // ── MIRDC Mail Middleware (PRODUCTION-VERIFIED, ported from RMISv4.1) ──
   106→      // POST {EMAIL_MIDDLEWARE_URL}/sendMail — Express+nodemailer service on
   107→      // the office VM. The From header is ALWAYS the office mailbox
   108→      // systems@mirdc.dost.gov.ph (display name = EMAIL_MIDDLEWARE_SYSTEM,
   109→      // default "DOST-MIRDC Recruitment"). Body carries the shared token as
   110→      // `api` (validated by the VM only when it opts into MIDDLEWARE_API_KEY
   111→      // strict mode), an HTML `message`, the plain-text part, and optional
   112→      // base64 attachments. Older middleware ignores text/attachments — this
   113→      // driver is safe against both versions.
   114→      try {
   115→        const message = input.bodyHtml ?? textToSimpleHtml(input.bodyText);
   116→        const res = await fetch(`${EMAIL_MIDDLEWARE_URL}/sendMail`, {
   117→          method: "POST",
   118→          headers: { "Content-Type": "application/json" },
   119→          // Never hang an API route on a stalled VM — the middleware normally
   120→          // answers in well under a second.
   121→          signal: AbortSignal.timeout(15_000),
   122→          body: JSON.stringify({
   123→            api: EMAIL_MIDDLEWARE_API_KEY,
   124→            system: EMAIL_MIDDLEWARE_SYSTEM || "DOST-MIRDC Recruitment",
   125→            // MOM 2026-10-13 — send as the recruitment mailbox. Middlewares
   126→            // that pin the From to the office mailbox ignore this field.
   127→            from: `"${RECRUITMENT_SENDER_NAME}" <${RECRUITMENT_EMAIL_ADDRESS}>`,
   128→            to: input.to,
   129→            subject: input.subject,
   130→            message,
   131→            text: input.bodyText,
   132→            ...(input.attachments && input.attachments.some((a) => a.content)
   133→              ? {
   134→                  attachments: input.attachments
   135→                    .filter((a) => a.content)
   136→                    .map((a) => ({
   137→                      filename: a.name,
   138→                      content: a.content,
   139→                      encoding: "base64",
   140→                      ...(a.contentType ? { contentType: a.contentType } : {}),
   141→                    })),
   142→                }
   143→              : {}),
   144→          }),
   145→        });
   146→        if (!res.ok) {
   147→          const text = await res.text().catch(() => "");
   148→          result = { status: "failed", error: `Middleware ${res.status}: ${text.slice(0, 200)}`, provider: "middleware" };
   149→        } else {
   150→          const json = (await res.json().catch(() => ({}))) as { messageId?: string | null };
   151→          result = { status: "sent", provider: "middleware", providerRef: json.messageId || "middleware" };
   152→        }
   153→      } catch (e) {
   154→        result = { status: "failed", error: e instanceof Error ? e.message : "middleware error", provider: "middleware" };
   155→      }
   156→    } else if (EMAIL_PROVIDER === "resend" && process.env.RESEND_API_KEY) {
   157→      // Real provider path (Resend HTTP API).
   158→      try {
   159→        const res = await fetch("https://api.resend.com/emails", {
   160→          method: "POST",
   161→          headers: {
   162→            Authorization: `Bearer ${process.env.RESEND_API_KEY}`,
   163→            "Content-Type": "application/json",
   164→          },
   165→          body: JSON.stringify({
   166→            from: process.env.EMAIL_FROM || `${RECRUITMENT_SENDER_NAME} <${RECRUITMENT_EMAIL_ADDRESS}>`,
   167→            to: [input.to],
   168→            subject: input.subject,
   169→            text: input.bodyText,
   170→            html: input.bodyHtml,
   171→          }),
   172→        });
   173→        if (!res.ok) {
   174→          const text = await res.text().catch(() => "");
   175→          result = { status: "failed", error: `Provider ${res.status}: ${text.slice(0, 200)}`, provider: "resend" };
   176→        } else {
   177→          const json = (await res.json()) as { id?: string };
   178→          result = { status: "sent", provider: "resend", providerRef: json.id ? String(json.id) : undefined };
   179→        }
   180→      } catch (e) {
   181→        result = { status: "failed", error: e instanceof Error ? e.message : "provider error", provider: "resend" };
   182→      }
   183→    } else if (EMAIL_PROVIDER === "gateway" && process.env.EMAIL_GATEWAY_URL) {
   184→      // mirdc_middleware email gateway (Task 28 contract): POST /api/rmis/send
   185→      // with X-RMIS-Key / Bearer authentication. The middleware owns SMTP
   186→      // credentials and relays delivery logs to the Strapi audit store.
   187→      try {
   188→        const base = process.env.EMAIL_GATEWAY_URL.replace(/\/+$/, "");
   189→        const res = await fetch(`${base}/api/rmis/send`, {
   190→          method: "POST",
   191→          headers: {
   192→            "Content-Type": "application/json",
   193→            "X-RMIS-Key": process.env.EMAIL_GATEWAY_KEY || "",
   194→            Authorization: `Bearer ${process.env.EMAIL_GATEWAY_KEY || ""}`,
   195→          },
   196→          body: JSON.stringify({
   197→            to: input.to,
   198→            subject: input.subject,
   199→            text: input.bodyText,
   200→            html: input.bodyHtml ?? input.bodyText,
   201→          }),
   202→        });
   203→        if (!res.ok) {
   204→          const text = await res.text().catch(() => "");
   205→          result = { status: "failed", error: `Gateway ${res.status}: ${text.slice(0, 200)}`, provider: "gateway" };
   206→        } else {
   207→          const json = (await res.json().catch(() => ({}))) as { id?: string; messageId?: string };
   208→          result = { status: "sent", provider: "gateway", providerRef: json.messageId ?? json.id ?? undefined };
   209→        }
   210→      } catch (e) {
   211→        result = { status: "failed", error: e instanceof Error ? e.message : "gateway error", provider: "gateway" };
   212→      }
   213→    } else {
   214→      result = { status: "mock", provider: EMAIL_PROVIDER };
   215→    }
   216→  } catch (e) {
   217→    result = { status: "failed", error: e instanceof Error ? e.message : "unknown", provider: EMAIL_PROVIDER };
   218→  }
   219→  try {
   220→    await db.emailLog.create({
   221→      data: {
   222→        to: input.to || "",
   223→        subject: input.subject,
   224→        bodyText: input.bodyText,
   225→        bodyHtml: input.bodyHtml ?? null,
   226→        provider: result.provider,
   227→        status: result.status,
   228→        providerRef: result.providerRef ?? null,
   229→        error: result.error ?? null,
   230→        relatedType: input.relatedType ?? null,
   231→        relatedId: input.relatedId ?? null,
   232→        // Metadata (name + size) ONLY — base64 bytes never touch the DB.
   233→        attachments: input.attachments
   234→          ? JSON.stringify(input.attachments.map(({ name, bytes }) => ({ name, bytes })))
   235→          : null,
   236→      },
   237→    });
   238→  } catch {
   239→    // log write is best-effort
   240→  }
   241→  return result;
   242→}

   244→export async function sendSms(input: { to: string; message: string; relatedType?: string; relatedId?: number }): Promise<SendResult> {
   245→  let result: SendResult;
   246→  try {
   247→    const normalized = normalizePhMobile(input.to);
   248→    if (!normalized) {
   249→      result = { status: "skipped", error: "Invalid Philippine mobile number", provider: SMS_PROVIDER };
   250→    } else if (SMS_PROVIDER === "semaphore" && process.env.SEMAPHORE_API_KEY) {
   251→      try {
   252→        const body = new URLSearchParams({
   253→          apikey: process.env.SEMAPHORE_API_KEY,
   254→          number: normalized.local,
   255→          message: input.message.slice(0, 640),
   256→          sendername: process.env.SEMAPHORE_SENDER || "MIRDCRMIS",
   257→        });
   258→        const res = await fetch("https://api.semaphore.co/api/v4/messages", {
   259→          method: "POST",
   260→          headers: { "Content-Type": "application/x-www-form-urlencoded" },
   261→          body,
   262→        });
   263→        result = res.ok
   264→          ? { status: "sent", provider: "semaphore" }
   265→          : { status: "failed", error: `Provider ${res.status}`, provider: "semaphore" };
   266→      } catch (e) {
   267→        result = { status: "failed", error: e instanceof Error ? e.message : "provider error", provider: "semaphore" };
   268→      }
   269→    } else {
   270→      result = { status: "mock", provider: SMS_PROVIDER };
   271→    }
   272→  } catch (e) {
   273→    result = { status: "failed", error: e instanceof Error ? e.message : "unknown", provider: SMS_PROVIDER };
   274→  }
   275→  try {
   276→    await db.smsLog.create({
   277→      data: {
   278→        to: input.to || "",
   279→        message: input.message,
   280→        provider: result.provider,
   281→        status: result.status,
   282→        providerRef: result.providerRef ?? null,
   283→        error: result.error ?? null,
   284→        relatedType: input.relatedType ?? null,
   285→        relatedId: input.relatedId ?? null,
   286→      },
   287→    });
   288→  } catch {
   289→    // best-effort
   290→  }
   291→  return result;
   292→}

   294→/** PH mobile normalization (spec §9.4): 0917…/+63917…/63917…/917… → local + E.164. */
   295→export function normalizePhMobile(raw: string): { local: string; e164: string } | null {
   296→  const digits = (raw || "").replace(/\D/g, "");
   297→  let local: string | null = null;
   298→  if (/^09\d{9}$/.test(digits)) local = digits;
   299→  else if (/^639\d{9}$/.test(digits)) local = `0${digits.slice(2)}`;
   300→  else if (/^9\d{9}$/.test(digits)) local = `0${digits}`;
   301→  if (!local) return null;
   302→  return { local, e164: `+63${local.slice(1)}` };
   303→}

   307→function escHtml(s: string): string {
   308→  return s
   309→    .replace(/&/g, "&amp;")
   310→    .replace(/</g, "&lt;")
   311→    .replace(/>/g, "&gt;")
   312→    .replace(/"/g, "&quot;");
   313→}

   315→/** Minimal escaped-HTML wrapper for plain-text bodies (middleware has no text-only mode). */
   316→function textToSimpleHtml(text: string): string {
   317→  const paras = text
   318→    .split(/\n{2,}/)
   319→    .map((p) => `<p style="margin:0 0 12px;">${escHtml(p).replace(/\n/g, "<br/>")}</p>`)
   320→    .join("");
   321→  return `<div style="font-family:Arial,Helvetica,sans-serif;font-size:14px;line-height:1.6;color:#1a1a1a;">${paras}</div>`;
   322→}

   419→export type NoticeDetails = {

   430→function noticeScheduleLines(d: NoticeDetails, kindLabel: string): string[] {
   431→  const online = d.mode === "online";
   432→  const lines = [
   433→    `Date: ${d.date || "TBA"}`,
   434→    `Time: ${d.time || "TBA"}`,
   435→    `${kindLabel} Mode: ${online ? "Online" : "Face-to-Face"}`,
   436→  ];
   437→  if (online) {
   438→    if (d.meetingLink) lines.push(`Meeting Link: ${d.meetingLink}`);
   439→    if (d.venue) lines.push(`Venue: ${d.venue}`);
   440→  } else {
   441→    lines.push(`Venue: ${d.venue || "TBA"}`);
   442→  }
   443→  if (d.contact) lines.push(`HR Contact: ${d.contact}`);
   444→  return lines;
   445→}
