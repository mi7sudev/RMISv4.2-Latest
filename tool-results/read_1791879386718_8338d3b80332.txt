   322→   322→export default function PersonalInfoSection({
   323→   323→  profile,
   324→   324→  seed,
   325→   325→  refreshSession,
   326→   326→}: {
   327→   327→  profile: ApplicantProfile | null;
   328→   328→  /** Bump to force a re-seed from the server profile (after PDS auto-apply / clear). */
   329→   329→  seed: number;
   330→   330→  refreshSession: () => Promise<unknown>;
   331→   331→}) {
   332→   332→  const [form, setForm] = useState<PersonalForm>(EMPTY_FORM);
   333→   333→  const [refs, setRefs] = useState<CharacterReference[]>([{ ...EMPTY_REF }]);
   334→   334→  const [saveState, setSaveState] = useState<"idle" | "dirty" | "saving" | "saved" | "error">("idle");
   335→   335→  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   336→   336→  const formRef = useRef(form);
   337→   337→  const refsRef = useRef(refs);
   338→   338→  const seededToken = useRef<string>("");
   339→   339→
   340→   340→  // Keep latest-value refs in sync after commit (never during render).
   341→   341→  useEffect(() => {
   342→   342→    formRef.current = form;
   343→   343→    refsRef.current = refs;
   344→   344→  }, [form, refs]);
   345→   345→
   346→   346→  // Seed (and re-seed on `seed` bumps) from the loaded profile — never during
   347→   347→  // silent reloads, so in-flight typing/autosave is never clobbered.
   348→   348→  useEffect(() => {
   349→   349→    if (!profile) return;
   350→   350→    const token = `${seed}`;
   351→   351→    if (seededToken.current === token) return;
   352→   352→    seededToken.current = token;
   353→   353→    setForm(seedForm(profile));
   354→   354→    const list = Array.isArray(profile.characterReferences) ? profile.characterReferences : [];
   355→   355→    setRefs(list.length > 0 ? list.map((r) => ({ ...EMPTY_REF, ...r })) : [{ ...EMPTY_REF }]);
   356→   356→    setSaveState("idle");
   357→   357→  }, [profile, seed]);
   358→   358→
   359→   359→  // Task 32 — official SignUpForm timing: rules are derived on EVERY render
   360→   360→  // (an error clears the moment its field is fixed); `shown` reveals a rule
   361→   361→  // only once the field has been left (onBlur) or Save Changes was pressed.
   362→   362→  // Typed over ALL PersonalForm keys (not just the ones with rules) so the
   363→   363→  // blur touch covers every field in the section.
   364→   364→  const rules: Partial<Record<keyof PersonalForm, string>> = {
   365→   365→    firstName: form.firstName.trim() === "" ? "First Name is required." : undefined,
   366→   366→    lastName: form.lastName.trim() === "" ? "Last Name is required." : undefined,
   367→   367→    emailAddress: form.emailAddress.trim() === "" ? "Email Address is required." : undefined,
   368→   368→    ...(form.adminCase && form.adminCaseDetails.trim() === ""
   369→   369→      ? { adminCaseDetails: "Describe the administrative case to continue." }
   370→   370→      : {}),
   371→   371→  };
   372→   372→  const { shown, touch, touchAll, hasGaps } = useAuthErrors(rules);
   373→   373→
   374→   374→  const save = useCallback(
   375→   375→    async (manual: boolean) => {
   376→   376→      if (timerRef.current) {
   377→   377→        clearTimeout(timerRef.current);
   378→   378→        timerRef.current = null;
   379→   379→      }
   380→   380→      setSaveState("saving");
   381→   381→      try {
   382→   382→        const payload: Record<string, string | boolean | CharacterReference[]> = {
   383→   383→          ...formRef.current,
   384→   384→          characterReferences: refsRef.current
   385→   385→            .map((r) => ({
   386→   386→              name: (r.name ?? "").trim(),
   387→   387→              title: (r.title ?? "").trim(),
   388→   388→              company: (r.company ?? "").trim(),
   389→   389→              companyAddress: (r.companyAddress ?? "").trim(),
   390→   390→              email: (r.email ?? "").trim(),
   391→   391→              contact: (r.contact ?? "").trim(),
   392→   392→            }))
   393→   393→            .filter((r) => r.name !== ""), // empty-name rows filtered at save (§7.4)
   394→   394→        };
   395→   395→        await apiFetch("/api/applicant/profile", { method: "PUT", body: payload });
   396→   396→        setSaveState("saved");
   397→   397→        if (manual) {
   398→   398→          toast.success("Profile changes saved");
   399→   399→          await refreshSession();
   400→   400→        }
   401→   401→      } catch (e) {
   402→   402→        setSaveState("error");
   403→   403→        if (manual) toast.error(e instanceof Error ? e.message : "Save failed. Please try again");
   404→   404→      }
   405→   405→    },
   406→   406→    [refreshSession]
   407→   407→  );
   408→   408→
   409→   409→  /** Re-arm the 1.2 s autosave debounce on every change (§7.4). Never
   410→   410→   *  triggered by the Task 32 error touch — only real edits save. */
   411→   411→  const armAutosave = useCallback(() => {
   412→   412→    setSaveState("dirty");
   413→   413→    if (timerRef.current) clearTimeout(timerRef.current);
   414→   414→    timerRef.current = setTimeout(() => {
   415→   415→      timerRef.current = null;
   416→   416→      void save(false);
   417→   417→    }, 1200);
   418→   418→  }, [save]);
   419→   419→
   420→   420→  useEffect(
   421→   421→    () => () => {
   422→   422→      if (timerRef.current) clearTimeout(timerRef.current);
   423→   423→    },
   424→   424→    []
   425→   425→  );
   426→   426→
   427→   427→  const set = useCallback(
   428→   428→    <K extends keyof PersonalForm>(key: K, value: PersonalForm[K]) => {
   429→   429→      setForm((prev) => ({ ...prev, [key]: value }));
   430→   430→      armAutosave();
   431→   431→    },
   432→   432→    [armAutosave]
   433→   433→  );
   434→   434→
   435→   435→  /** Manual save — Task 32 official SignUpForm timing: Save Changes reveals
   436→   436→   *  every gap at once (touchAll) and the save only proceeds when the live
   437→   437→   *  rules are clean; correcting a field clears its rule instantly. */
   438→   438→  const saveManual = useCallback(() => {
   439→   439→    touchAll();
   440→   440→    if (hasGaps) return;
   441→   441→    void save(true);
